cmake_minimum_required(VERSION 3.13)
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(robot_controller C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT ROBOT_GENERATED_DIR)
  message(FATAL_ERROR "Generate a per-robot joint map with tools/build_firmware.py")
endif()
add_executable(robot_controller target_main.c wire.c actuator.c supervisor.c imu.c)
target_include_directories(robot_controller PRIVATE ${ROBOT_GENERATED_DIR} ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(robot_controller PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(robot_controller PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=0)
target_link_libraries(robot_controller pico_stdlib pico_rand hardware_uart hardware_adc hardware_watchdog hardware_i2c)
pico_enable_stdio_usb(robot_controller 1)
pico_enable_stdio_uart(robot_controller 0)
pico_add_extra_outputs(robot_controller)
